def move_up(state):
    row, col = find_blank(state)
    if row == 0:
        return None
    new_state = [r.copy() for r in state]
    new_state[row][col] = new_state[row-1][col]
    new_state[row-1][col] = 0
    return new_state

def move_down(state):
    row, col = find_blank(state)
    if row == 2:
        return None
    new_state = [r.copy() for r in state]
    new_state[row][col] = new_state[row+1][col]
    new_state[row+1][col] = 0
    return new_state
